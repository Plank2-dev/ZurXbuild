name: Unzip and commit MR_PLANKTON_ZURXBUILD

on:
  workflow_dispatch: {}

permissions:
  contents: write

jobs:
  unzip-and-commit:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: Ensure unzip is available
        run: sudo apt-get update && sudo apt-get install -y unzip

      - name: Unzip and commit files
        run: |
          ZIP=MR_PLANKTON_ZURXBUILD_GITHUB_UPLOAD.zip
          if [ ! -f "$ZIP" ]; then
            echo "File $ZIP not found in repository root."
            exit 1
          fi

          # Extract to temporary folder to avoid touching .git
          TMPDIR=$(mktemp -d)
          unzip -o "$ZIP" -d "$TMPDIR"

          # Move extracted files into repo root (including dotfiles)
          shopt -s dotglob || true
          # Copy contents, but keep .git
          rsync -a --exclude='.git' "$TMPDIR/" ./

          # Clean up
          rm -rf "$TMPDIR"

          # Commit and push
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add -A
          git commit -m "Unzip MR_PLANKTON_ZURXBUILD_GITHUB_UPLOAD.zip and add extracted files" || echo "No changes to commit"
          git push
